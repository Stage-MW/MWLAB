function simulateAA() { const aaKey = document.getElementById('sim-aa-select').value; const misKey = document.getElementById('sim-missile-select').value; const aaQty = parseInt(document.getElementById('sim-aa-qty').value); const misQty = parseInt(document.getElementById('sim-missile-qty').value); const distance = parseFloat(document.getElementById('sim-dist').value); if (!aaKey || !misKey) return showToast("ОШИБКА: ОРУЖИЕ НЕ ВЫБРАНО", true); const aa = DB[aaKey]; const mis = DB[misKey]; // 1. Физика полета ракеты (как было) const MW_COEF = 0.00278; let v0 = (mis.start_speed || mis.max_speed || 0) * MW_COEF; let vmax = (mis.max_speed || 0) * MW_COEF; let accelTime = mis.accel_time || 0.1; let accelDist = ((v0 + vmax) / 2) * accelTime; let flyTime = 0; if (distance <= accelDist) { let currentSpeed = v0 + ((vmax - v0) * (distance / accelDist)); flyTime = distance / ((v0 + currentSpeed) / 2); } else { flyTime = accelTime + ((distance - accelDist) / vmax); } // 2. Расчет времени в зоне поражения ПВО (как было) let aaRange = aa.range || 3.0; let timeToEnterAARange = 0; if (distance > aaRange) { let distToAa = distance - aaRange; if (distToAa <= accelDist) { let speedAtAa = v0 + ((vmax - v0) * (distToAa / accelDist)); timeToEnterAARange = distToAa / ((v0 + speedAtAa) / 2); } else { timeToEnterAARange = accelTime + ((distToAa - accelDist) / vmax); } } let engageTime = Math.max(0, flyTime - timeToEnterAARange); // 3. Формирование ракетного роя let projectilesPerLaunch = mis.projectiles || 1; let totalMissiles = misQty * projectilesPerLaunch; let singleMissileHP = mis.hp || 50; let totalMissileHP = singleMissileHP * totalMissiles; // 4. НОВЫЙ АЛГОРИТМ УРОНА ПВО С УЧЕТОМ ЗАДЕРЖКИ let ammoClip = aa.ammo_clip || 1; let fireDelay = aa.fire_delay || (aa.aa_type === "Ракетное" ? 0.25 : 0.15); // Фоллбэк, если параметр не найден // Если выстрелов больше 1, считаем время непрерывного огня let burstDuration = (ammoClip > 1) ? (ammoClip * fireDelay) : fireDelay; // Используем effectiveBurst, который симулятор рассчитывает при загрузке (с учетом точности) let damageBurst = aa.effectiveBurst || aa.damage_burst || 0; // Урон в секунду В МОМЕНТ ЗАЛПА let burstDps = burstDuration > 0 ? (damageBurst / burstDuration) : damageBurst; let reloadTime = aa.reload || 3.0; let cycleTime = burstDuration + reloadTime; // Сколько полных циклов (залп + перезарядка) успеет сделать ПВО let fullCycles = Math.floor(engageTime / cycleTime); let remainderTime = engageTime - (fullCycles * cycleTime); // Урон за остаток времени (до ухода на очередную перезарядку) let remainderDamage = 0; if (burstDuration > 0) { // ПВО успеет нанести только часть урона из обоймы remainderDamage = Math.min(damageBurst, burstDps * remainderTime); } else { remainderDamage = remainderTime > 0 ? damageBurst : 0; } let damagePerModule = (fullCycles * damageBurst) + remainderDamage; let totalAADamage = damagePerModule * aaQty; // 5. Итоги перехвата let destroyedMissiles = Math.floor(totalAADamage / singleMissileHP); destroyedMissiles = Math.min(destroyedMissiles, totalMissiles); let survivedMissiles = totalMissiles - destroyedMissiles; let survivalProbability = (survivedMissiles / totalMissiles) * 100; let resColor = survivalProbability > 50 ? 'var(--secondary)' : 'var(--primary)'; let verdict = survivalProbability === 0 ? "ПОЛНЫЙ ПЕРЕХВАТ ЗАЛПА" : "ПРОРЫВ ПВО УСПЕШЕН!"; const resDiv = document.getElementById('sim-result'); resDiv.style.display = 'block'; resDiv.innerHTML = `
Время реакции ПВО: ${engageTime.toFixed(2)} сек.
Математический урон ПВО: ${Math.round(totalAADamage).toLocaleString('ru')} HP
Прочность роя ракет (${totalMissiles} шт.): ${Math.round(totalMissileHP).toLocaleString('ru')} HP
${verdict}
Сбито ракет: ${destroyedMissiles} из ${totalMissiles}
Проходимость урона: ${survivalProbability.toFixed(1)}%
`; showToast("СИМУЛЯЦИЯ ЗАВЕРШЕНА"); }
